# Run the collector once a day at 06:15 local time (crontab -e). Adjust the repo path.
# Put PHISHTANK_APP_KEY / URLHAUS_AUTH_KEY in ~/.phishdrift.env (never commit it).
15 6 * * * cd $HOME/phishdrift && set -a && . $HOME/.phishdrift.env && set +a && .venv/bin/python -m collector run >> data/collector.log 2>&1
